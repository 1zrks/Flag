-- GHOST LIMB EXTENDER - STANDALONE
-- ============================================

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local localPlayer = Players.LocalPlayer

local DEFAULTS = {
    TOGGLE_KEY = "L",
    TARGET_LIMB = "HumanoidRootPart",
    LIMB_SIZE = 15,
    LIMB_TRANSPARENCY = 0.8,
    LIMB_CAN_COLLIDE = false,
    TEAM_CHECK = true,
    USE_HIGHLIGHT = true,
    HIGHLIGHT_FILL_COLOR = Color3.fromRGB(255, 117, 24),
    HIGHLIGHT_FILL_TRANSPARENCY = 0.5,
    HIGHLIGHT_OUTLINE_TRANSPARENCY = 1,
}

-- ============================================
-- CLEANUP OLD INSTANCE
-- ============================================
if getgenv().GhostLimbExtender_Cleanup and type(getgenv().GhostLimbExtender_Cleanup) == "function" then
    pcall(getgenv().GhostLimbExtender_Cleanup)
end

-- ============================================
-- CONNECTION MANAGER (Lightweight, self-contained)
-- ============================================
local ConnectionManager = {}
ConnectionManager.__index = ConnectionManager

function ConnectionManager.new()
    return setmetatable({ _connections = {} }, ConnectionManager)
end

function ConnectionManager:Connect(signal, callback)
    if not signal or not callback then return end
    local conn = signal:Connect(callback)
    table.insert(self._connections, conn)
    return conn
end

function ConnectionManager:DisconnectAll()
    for _, conn in ipairs(self._connections) do
        pcall(function() conn:Disconnect() end)
    end
    self._connections = {}
end

function ConnectionManager:Destroy()
    self:DisconnectAll()
end

-- ============================================
-- GHOST PART CREATION
-- ============================================
local function createGhostPart(originalPart, settings)
    local ghost = Instance.new("Part")
    ghost.Name = "Handle"
    ghost.Size = Vector3.new(settings.LIMB_SIZE, settings.LIMB_SIZE, settings.LIMB_SIZE)
    ghost.Transparency = settings.LIMB_TRANSPARENCY
    ghost.CanCollide = settings.LIMB_CAN_COLLIDE
    ghost.Anchored = false
    ghost.Massless = true
    ghost.CastShadow = false
    ghost.CanQuery = false
    ghost.CanTouch = false

    pcall(function()
        ghost.Color = originalPart.Color
        ghost.Material = originalPart.Material
    end)

    ghost.Parent = nil -- GHOST MODE: invisible to workspace enumeration

    local weld = Instance.new("WeldConstraint")
    weld.Part0 = ghost
    weld.Part1 = originalPart
    weld.Parent = ghost

    -- Hide the original part so only the ghost shows
    originalPart.Transparency = 1

    local highlight = nil
    if settings.USE_HIGHLIGHT then
        highlight = Instance.new("Highlight")
        highlight.FillColor = settings.HIGHLIGHT_FILL_COLOR or Color3.fromRGB(255, 117, 24)
        highlight.FillTransparency = settings.HIGHLIGHT_FILL_TRANSPARENCY or 0.5
        highlight.OutlineTransparency = settings.HIGHLIGHT_OUTLINE_TRANSPARENCY or 1
        highlight.Adornee = ghost
        highlight.Enabled = true
        highlight.Parent = ghost
    end

    return ghost, weld, highlight
end

local function destroyGhostPart(ghost, originalPart)
    if originalPart and originalPart.Parent then
        originalPart.Transparency = 0
    end
    if ghost then
        ghost:Destroy()
    end
end

-- ============================================
-- PLAYER DATA (Manages one ghost per player)
-- ============================================
local PlayerData = {}
PlayerData.__index = PlayerData

function PlayerData.new(controller, player)
    local self = setmetatable({
        _controller = controller,
        player = player,
        _ghostData = nil, -- { ghost, highlight, originalPart }
        _conns = ConnectionManager.new(),
        _destroyed = false,
    }, PlayerData)

    if player.CharacterAdded then
        self._conns:Connect(player.CharacterAdded, function(char)
            if not self._destroyed then
                self:onCharacter(char)
            end
        end)
    end

    local char = player.Character or workspace:FindFirstChild(player.Name)
    if char then
        self:onCharacter(char)
    end

    return self
end

function PlayerData:attachGhost(targetPart)
    if self._ghostData then return end -- Already attached

    local settings = self._controller._settings
    local ghost, _, highlight = createGhostPart(targetPart, settings)
    self._ghostData = {
        ghostPart = ghost,
        highlight = highlight,
        originalPart = targetPart,
    }

    -- Cleanup on death
    local char = targetPart.Parent
    if char then
        local humanoid = char:FindFirstChildOfClass("Humanoid")
        if humanoid and humanoid.Died then
            self._conns:Connect(humanoid.Died, function()
                self:detachGhost()
            end)
        end
    end

    -- Cleanup on character removal
    if self.player.CharacterRemoving then
        self._conns:Connect(self.player.CharacterRemoving, function()
            self:detachGhost()
        end)
    end
end

function PlayerData:detachGhost()
    if not self._ghostData then return end
    destroyGhostPart(self._ghostData.ghostPart, self._ghostData.originalPart)
    if self._ghostData.highlight then
        pcall(function() self._ghostData.highlight:Destroy() end)
    end
    self._ghostData = nil
end

function PlayerData:onCharacter(char)
    if not char or self._destroyed then return end
    if self._controller:_isTeam(self.player) then return end

    local humanoid = char:FindFirstChildOfClass("Humanoid")
    if not humanoid or humanoid.Health <= 0 then return end

    local targetPart = char:FindFirstChild(self._controller._settings.TARGET_LIMB)
    if not targetPart or not targetPart:IsA("BasePart") then return end

    self:detachGhost()
    self:attachGhost(targetPart)
end

function PlayerData:Destroy()
    if self._destroyed then return end
    self._destroyed = true

    self:detachGhost()
    self._conns:Destroy()

    setmetatable(self, nil)
    for k in pairs(self) do self[k] = nil end
end

-- ============================================
-- MAIN CONTROLLER
-- ============================================
local GhostLimbExtender = {}
GhostLimbExtender.__index = GhostLimbExtender

function GhostLimbExtender.new(userSettings)
    local self = setmetatable({
        _settings = {},
        _playerData = {},
        _conns = ConnectionManager.new(),
        _running = false,
        _destroyed = false,
    }, GhostLimbExtender)

    -- Merge settings
    for k, v in pairs(DEFAULTS) do self._settings[k] = v end
    if userSettings then
        for k, v in pairs(userSettings) do self._settings[k] = v end
    end

    -- Keybind via UserInputService (no external deps)
    local uis = game:GetService("UserInputService")
    self._conns:Connect(uis.InputBegan, function(input, gameProcessed)
        if gameProcessed then return end
        if input.KeyCode == Enum.KeyCode[self._settings.TOGGLE_KEY] then
            self:Toggle()
        end
    end)

    -- Mobile button via ScreenGui
    local screenGui = Instance.new("ScreenGui")
    screenGui.Name = "GhostLimbExtender"
    screenGui.Parent = localPlayer:WaitForChild("PlayerGui")

    local toggleButton = Instance.new("TextButton")
    toggleButton.Size = UDim2.new(0, 140, 0, 35)
    toggleButton.Position = UDim2.new(0, 5, 0, 5)
    toggleButton.Text = "GHOST: OFF"
    toggleButton.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
    toggleButton.TextColor3 = Color3.fromRGB(200, 200, 200)
    toggleButton.Font = Enum.Font.Code
    toggleButton.TextSize = 12
    toggleButton.Parent = screenGui

    self._toggleButton = toggleButton

    toggleButton.MouseButton1Click:Connect(function()
        self:Toggle()
    end)

    -- Mobile drag support
    local dragging = false
    local dragStart = nil
    local startPos = nil

    toggleButton.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true
            dragStart = input.Position
            startPos = toggleButton.Position
        end
    end)

    uis.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseMovement) then
            local delta = input.Position - dragStart
            toggleButton.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)

    uis.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
        end
    end)

    -- Register global cleanup
    getgenv().GhostLimbExtender_Cleanup = function()
        pcall(function() self:Destroy() end)
    end

    return self
end

function GhostLimbExtender:_isTeam(player)
    return self._settings.TEAM_CHECK
        and localPlayer
        and localPlayer.Team ~= nil
        and player.Team == localPlayer.Team
end

function GhostLimbExtender:Start()
    if self._running then return end
    self._running = true

    if self._toggleButton then
        self._toggleButton.Text = "GHOST: ON"
        self._toggleButton.BackgroundColor3 = Color3.fromRGB(80, 20, 20)
    end

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= localPlayer then
            if not self._playerData[player.Name] then
                self._playerData[player.Name] = PlayerData.new(self, player)
            end
        end
    end

    self._conns:Connect(Players.PlayerAdded, function(player)
        if not self._playerData[player.Name] then
            self._playerData[player.Name] = PlayerData.new(self, player)
        end
    end)

    self._conns:Connect(Players.PlayerRemoving, function(player)
        local pd = self._playerData[player.Name]
        if pd then
            pd:Destroy()
            self._playerData[player.Name] = nil
        end
    end)
end

function GhostLimbExtender:Stop()
    if not self._running then return end
    self._running = false

    if self._toggleButton then
        self._toggleButton.Text = "GHOST: OFF"
        self._toggleButton.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
    end

    self._conns:DisconnectAll()

    for _, pd in pairs(self._playerData) do
        pd:Destroy()
    end
    self._playerData = {}
end

function GhostLimbExtender:Toggle()
    if self._running then
        self:Stop()
    else
        self:Start()
    end
end

function GhostLimbExtender:Destroy()
    if self._destroyed then return end
    self._destroyed = true

    self:Stop()
    self._conns:Destroy()

    if self._toggleButton then
        self._toggleButton:Destroy()
        self._toggleButton = nil
    end

    getgenv().GhostLimbExtender_Cleanup = nil
end
