-- ============================================
-- GHOST LIMB EXTENDER - FINAL STEALTH BUILD
-- ============================================

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local localPlayer = Players.LocalPlayer

local DEFAULTS = {
    TOGGLE = "L",
    TARGET_LIMB = "HumanoidRootPart",
    LIMB_SIZE = 15,
    LIMB_TRANSPARENCY = 0.9,
    LIMB_CAN_COLLIDE = false,
    MOBILE_BUTTON = true,
    LISTEN_FOR_INPUT = true,
    TEAM_CHECK = true,
    FORCEFIELD_CHECK = true,
    RESET_LIMB_ON_DEATH2 = false,
    USE_HIGHLIGHT = true,
    DEPTH_MODE = "AlwaysOnTop",
    HIGHLIGHT_FILL_COLOR = Color3.fromRGB(255, 117, 24),
    HIGHLIGHT_FILL_TRANSPARENCY = 0.7,
    HIGHLIGHT_OUTLINE_COLOR = Color3.fromRGB(0, 0, 0),
    HIGHLIGHT_OUTLINE_TRANSPARENCY = 1,
}

local limbExtenderData = getgenv().limbExtenderData or {}
getgenv().limbExtenderData = limbExtenderData

-- Destroy previous instance cleanly
if limbExtenderData._cleanup and type(limbExtenderData._cleanup) == "function" then
    pcall(limbExtenderData._cleanup)
end

-- ============================================
-- EXTERNAL DEPENDENCIES (Load quietly)
-- ============================================
local function safeLoad(url)
    local success, result = pcall(function()
        return loadstring(game:HttpGet(url))()
    end)
    return success and result or nil
end

local ConnectionManager = limbExtenderData.ConnectionManager or safeLoad('https://raw.githubusercontent.com/AAPVdev/modules/refs/heads/main/ConnectionManager.lua')
local ContextActionUtility = limbExtenderData.ContextActionUtility or safeLoad('https://raw.githubusercontent.com/AAPVdev/modules/refs/heads/main/ContextActionUtility.lua')

limbExtenderData.ConnectionManager = ConnectionManager
limbExtenderData.ContextActionUtility = ContextActionUtility

-- ============================================
-- DECOY FOLDER (Empty, for misdirection)
-- ============================================
local decoyFolder = nil
local function getDecoyFolder()
    if decoyFolder and decoyFolder.Parent then
        return decoyFolder
    end
    decoyFolder = Instance.new("Folder")
    decoyFolder.Name = "LimbExtenderClones"
    decoyFolder.Parent = workspace
    return decoyFolder
end

-- ============================================
-- GHOST PART CREATOR
-- ============================================
local function createGhostPart(originalPart, settings)
    local sizeVal = settings.LIMB_SIZE or DEFAULTS.LIMB_SIZE
    
    -- The ghost: parented to nil, named to blend in
    local ghostPart = Instance.new("Part")
    ghostPart.Name = "Handle" -- Blends in with standard accessory handles
    ghostPart.Size = Vector3.new(sizeVal, sizeVal, sizeVal)
    ghostPart.Transparency = settings.LIMB_TRANSPARENCY
    ghostPart.CanCollide = settings.LIMB_CAN_COLLIDE
    ghostPart.Anchored = false
    ghostPart.Massless = true
    ghostPart.CastShadow = false
    ghostPart.CanQuery = false
    ghostPart.CanTouch = false
    
    -- Make it look like the target to help the highlight blend
    pcall(function()
        ghostPart.Color = originalPart.Color
        ghostPart.Material = originalPart.Material
    end)

    -- CRITICAL: Parent to nil so it does not appear in Workspace
    ghostPart.Parent = nil
    
    -- Weld the ghost (parented to nil) to the real target (in Workspace)
    local weld = Instance.new("WeldConstraint")
    weld.Part0 = ghostPart
    weld.Part1 = originalPart
    weld.Parent = ghostPart -- The weld is parented to the ghost, keeping everything hidden

    -- Hide the original part to make the ghost the only visible/interactable piece
    originalPart.Transparency = 1
    
    return ghostPart, weld
end

local function destroyGhostPart(ghostPart, originalPart)
    -- Restore the original part's visibility
    if originalPart and originalPart.Parent then
        originalPart.Transparency = 0
    end
    -- The weld is parented to the ghost, so destroying the ghost destroys the weld too.
    if ghostPart then
        ghostPart:Destroy()
    end
end

-- ============================================
-- HIGHLIGHT (Attached to the ghost part)
-- ============================================
local function makeHighlight(ghostPart, settings)
    local hiFolder = game:GetService("ReplicatedStorage"):FindFirstChild("Limb Extender Highlights Folder")
    if not hiFolder then
        hiFolder = Instance.new("Folder")
        hiFolder.Name = "Limb Extender Highlights Folder"
        hiFolder.Parent = game:GetService("ReplicatedStorage")
    end
    local hi = Instance.new("Highlight")
    hi.Name = "LimbHighlight"
    
    pcall(function()
        if settings.DEPTH_MODE and Enum.HighlightDepthMode[settings.DEPTH_MODE] then
            hi.DepthMode = Enum.HighlightDepthMode[settings.DEPTH_MODE]
        end
        hi.FillColor = settings.HIGHLIGHT_FILL_COLOR or Color3.fromRGB(255, 117, 24)
        hi.FillTransparency = settings.HIGHLIGHT_FILL_TRANSPARENCY or 0.7
        hi.OutlineColor = settings.HIGHLIGHT_OUTLINE_COLOR or Color3.fromRGB(0, 0, 0)
        hi.OutlineTransparency = settings.HIGHLIGHT_OUTLINE_TRANSPARENCY or 1
        hi.Adornee = ghostPart
        hi.Enabled = true
    end)
    
    hi.Parent = hiFolder
    return hi
end

-- ============================================
-- PLAYER DATA (Manages the ghost parts for each player)
-- ============================================
local PlayerData = {}
PlayerData.__index = PlayerData

function PlayerData.new(parent, player)
    local self = setmetatable({
        _parent = parent,
        player = player,
        _ghosts = {}, -- { [originalPart] = { ghostPart = ghost, highlight = hi } }
        _charDelay = nil,
        _destroyed = false,
    }, PlayerData)

    -- Connect to character events
    if player and player.CharacterAdded then
        player.CharacterAdded:Connect(function(c) 
            if not self._destroyed then
                self:onCharacter(c) 
            end
        end)
    end
    
    -- Check if the player already has a character
    local character = player and (player.Character or workspace:FindFirstChild(player.Name))
    if character then
        self:onCharacter(character)
    end
    
    return self
end

function PlayerData:createGhost(part)
    if not part or self._ghosts[part] then return end -- Avoid duplicates
    
    local settings = self._parent._settings
    
    -- Create the ghost part and its highlight
    local ghostPart, _ = createGhostPart(part, settings)
    local highlight = nil
    if settings.USE_HIGHLIGHT then
        highlight = makeHighlight(ghostPart, settings)
    end
    
    -- Store everything for cleanup
    self._ghosts[part] = {
        ghostPart = ghostPart,
        highlight = highlight,
    }
end

function PlayerData:destroyGhost(part)
    local data = self._ghosts[part]
    if not data then return end
    
    if data.highlight then
        pcall(function() data.highlight:Destroy() end)
    end
    destroyGhostPart(data.ghostPart, part)
    self._ghosts[part] = nil
end

function PlayerData:destroyAllGhosts()
    for part, _ in pairs(self._ghosts) do
        self:destroyGhost(part)
    end
end

function PlayerData:setupCharacter(char)
    local parent = self._parent
    if not char or not parent or self._destroyed then return end
    if not self.player then return end

    -- Skip team members
    if parent:_isTeam(self.player) then return end

    -- Ensure character is alive
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    if not humanoid or humanoid.Health <= 0 then return end

    -- Find the target limb (e.g., HumanoidRootPart)
    local targetPart = char:FindFirstChild(parent._settings.TARGET_LIMB)
    if not targetPart or not targetPart:IsA("BasePart") then return end

    -- Create the ghost!
    self:createGhost(targetPart)

    -- Cleanup on death
    local deathEvent = parent._settings.RESET_LIMB_ON_DEATH2 and humanoid.HealthChanged or humanoid.Died
    if deathEvent then
        local deathConn
        deathConn = deathEvent:Connect(function(hp)
            if not hp or hp <= 0 then
                self:destroyGhost(targetPart)
                if deathConn then deathConn:Disconnect() end
            end
        end)
    end

    -- Cleanup when the character is removed
    if self.player and self.player.CharacterRemoving then
        local removeConn
        removeConn = self.player.CharacterRemoving:Connect(function()
            self:destroyAllGhosts()
            if removeConn then removeConn:Disconnect() end
        end)
    end
end

function PlayerData:onCharacter(char)
    if not char or self._destroyed then return end
    
    -- Cancel any pending setup
    if self._charDelay then 
        task.cancel(self._charDelay)
        self._charDelay = nil 
    end

    -- Delay setup slightly to avoid burst processing on spawn
    self._charDelay = task.delay(0.5, function()
        if self._destroyed then return end
        
        local parent = self._parent
        -- If the player has a forcefield, wait for it to drop
        if parent._settings.FORCEFIELD_CHECK then
            local ff = char:FindFirstChildOfClass("ForceField")
            if ff then
                local ffConn
                ffConn = ff.Destroying:Connect(function()
                    self:setupCharacter(char)
                    if ffConn then ffConn:Disconnect() end
                end)
                return
            end
        end
        self:setupCharacter(char)
    end)
end

function PlayerData:Destroy()
    if self._destroyed then return end
    self._destroyed = true

    -- Destroy all created ghosts
    self:destroyAllGhosts()

    -- Cancel pending operations
    if self._charDelay then 
        pcall(function() task.cancel(self._charDelay) end)
        self._charDelay = nil 
    end

    -- Prevent memory leaks
    setmetatable(self, nil)
    for k in pairs(self) do self[k] = nil end
end

-- ============================================
-- MAIN LIMB EXTENDER CONTROLLER
-- ============================================
local LimbExtender = {}
LimbExtender.__index = LimbExtender

function LimbExtender.new(userSettings)
    local self = setmetatable({
        _settings = mergeSettings(userSettings),
        _playerTable = {},
        _connections = {},
        _running = false,
        _destroyed = false,
    }, LimbExtender)

    -- Deferred startup to blend in
    task.delay(0.8, function()
        if self._destroyed then return end
        
        -- Bind the toggle key
        if self._settings.LISTEN_FOR_INPUT and ContextActionUtility and type(ContextActionUtility.BindAction) == "function" then
            ContextActionUtility:BindAction(
                "LimbExtenderToggle",
                function(_, inputState)
                    if inputState == Enum.UserInputState.Begin then
                        self:Toggle()
                    end
                end,
                self._settings.MOBILE_BUTTON,
                Enum.KeyCode[self._settings.TOGGLE]
            )
        end
        
        -- Create the empty decoy folder
        getDecoyFolder()
    end)

    -- Store cleanup function globally for re-injection handling
    limbExtenderData._cleanup = function()
        pcall(function() self:Destroy() end)
    end

    return self
end

function LimbExtender:_isTeam(player)
    return self._settings.TEAM_CHECK 
        and localPlayer 
        and localPlayer.Team ~= nil 
        and player.Team == localPlayer.Team
end

function LimbExtender:_connect(signal, callback)
    if not signal then return end
    local conn = signal:Connect(callback)
    table.insert(self._connections, conn)
    return conn
end

function LimbExtender:_disconnectAll()
    for _, conn in ipairs(self._connections) do
        pcall(function() conn:Disconnect() end)
    end
    self._connections = {}
end

function LimbExtender:Start()
    if self._running then return end
    self._running = true

    -- Create ghosts for all current players
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= localPlayer then
            if not self._playerTable[p.Name] then
                self._playerTable[p.Name] = PlayerData.new(self, p)
            end
        end
    end

    -- Restart on local player team change
    if localPlayer then
        local sig = localPlayer:GetPropertyChangedSignal("Team")
        if sig then
            self:_connect(sig, function() self:Restart() end)
        end
    end

    -- Handle new players joining
    self:_connect(Players.PlayerAdded, function(p)
        if not self._playerTable[p.Name] then
            self._playerTable[p.Name] = PlayerData.new(self, p)
        end
    end)

    -- Handle players leaving
    self:_connect(Players.PlayerRemoving, function(p)
        local pd = self._playerTable[p.Name]
        if pd and type(pd.Destroy) == "function" then 
            pd:Destroy() 
        end
        self._playerTable[p.Name] = nil
    end)

    -- Update mobile button UI
    if self._settings.MOBILE_BUTTON and ContextActionUtility and type(ContextActionUtility.SetTitle) == "function" then
        ContextActionUtility:SetTitle("LimbExtenderToggle", "On")
    end
end

function LimbExtender:Stop()
    if not self._running then return end
    self._running = false

    self:_disconnectAll()

    -- Destroy all player data and their ghosts
    for _, pd in pairs(self._playerTable) do
        if pd and type(pd.Destroy) == "function" then 
            pd:Destroy() 
        end
    end
    self._playerTable = {}

    -- Update mobile button UI
    if self._settings.MOBILE_BUTTON and ContextActionUtility and type(ContextActionUtility.SetTitle) == "function" then
        ContextActionUtility:SetTitle("LimbExtenderToggle", "Off")
    end
end

function LimbExtender:Toggle()
    if self._running then
        self:Stop()
    else
        self:Start()
    end
end

function LimbExtender:Restart()
    local wasRunning = self._running
    self:Stop()
    if wasRunning then 
        task.delay(0.3, function()
            if not self._destroyed then
                self:Start()
            end
        end)
    end
end

function LimbExtender:Destroy()
    if self._destroyed then return end
    self._destroyed = true

    self:Stop()

    -- Unbind key
    if ContextActionUtility and type(ContextActionUtility.UnbindAction) == "function" then
        pcall(function() ContextActionUtility:UnbindAction("LimbExtenderToggle") end)
    end

    -- Remove decoy folder
    if decoyFolder then
        pcall(function() decoyFolder:Destroy() end)
        decoyFolder = nil
    end

    limbExtenderData._cleanup = nil
end

function LimbExtender:Set(key, value)
    if self._settings[key] ~= value then
        self._settings[key] = value
        self:Restart()
    end
end

function LimbExtender:Get(key) 
    return self._settings[key] 
end

return setmetatable({}, { 
    __call = function(_, userSettings) 
        return LimbExtender.new(userSettings) 
    end 
})
